export function middleware(request: NextRequest) {
  if (process.env.AUTH_ENABLED === "0") return NextResponse.next(); // D-03 dev bypass
  const { pathname } = request.nextUrl;
  if (!request.cookies.has("qb_session")) {

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| browser -> middleware | Every /dash + /api/tenants request; only the cookie's presence is checked here |
| browser -> /dash/[id] page | URL `[id]` slug is attacker-controllable; must not be trusted for identity |
| browser -> /sign-in | `?next=` param crosses here |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-17 | Elevation of Privilege | unauthenticated access to tenant data | mitigate | `middleware.ts` coarse-gates `/dash` + `/api/tenants`; the dashboard page additionally calls `resolveTenant()` for full session validation (AUTH-06) |
| T-06-18 | Elevation of Privilege | user views another user's dashboard via URL slug | mitigate | `app/dash/[id]/page.tsx` ignores the URL slug for identity — `resolveTenant()` resolves the brain from the session; a slug mismatch redirects/404s |
| T-06-19 | Tampering | `postgres` import crashing Edge middleware | mitigate | `middleware.ts` does cookie-presence only; imports no `postgres`-touching module (Pitfall 2) |
| T-06-20 | Spoofing | open redirect via `?next=` on the sign-in page | mitigate | `?next=` carried opaquely, validated server-side by `isSafeNextPath` before any redirect honors it; never rendered as a visible link |
| T-06-21 | Information Disclosure | stale filesystem tenant registry leaking cross-tenant rows | mitigate | The registry is now Postgres-backed keyed on `app.users`; no shared in-memory `Map` rebuilt from a disk scan |
</threat_model>
