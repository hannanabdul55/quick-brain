DB_PATH="${REPO_ROOT}/data/quickbrain-app.sqlite"

FORCE_REAL=0
for arg in "$@"; do
  [ "$arg" = "--force-real-tenants" ] && FORCE_REAL=1
done

AUTH_ENABLED="${AUTH_ENABLED:-0}"

# AUTH-12: Gate — refuse to run against real users without explicit confirmation.
if [ "$AUTH_ENABLED" = "1" ] && [ $FORCE_REAL -eq 0 ]; then
  echo "[panic-reset] ERROR: AUTH_ENABLED=1 means real user accounts exist."
  echo "[panic-reset] Re-run with --force-real-tenants to confirm you want to wipe real user data."
  echo "[panic-reset] This will delete all user accounts and their associated brain directories."
  exit 1
fi

# AUTH-12: Enumerate affected users when forcing real-tenant wipe.
if [ "$AUTH_ENABLED" = "1" ] && [ $FORCE_REAL -eq 1 ]; then
  if command -v sqlite3 >/dev/null 2>&1 && [ -f "$DB_PATH" ]; then
    echo "[panic-reset] Real users that will lose access (brain dirs wiped, accounts remain in DB):"
    sqlite3 "$DB_PATH" "SELECT email FROM users;" 2>/dev/null | while read -r email; do
      echo "[panic-reset]   - $email"
    done
  else
    echo "[panic-reset] WARNING: Cannot enumerate users (sqlite3 not on PATH or DB not found at $DB_PATH)"
  fi
  echo ""
  read -r -p "[panic-reset] Type 'yes' to confirm deletion of all real tenant brain dirs: " confirm
  if [ "$confirm" != "yes" ]; then
    echo "[panic-reset] Aborted."
    exit 1
  fi
fi
