<objective>
Build the application-side gbrain spawn harness: a single `spawnGBrain(args, opts)` helper, per-tenant mutex queue, typed errors, slug validation, and a tenant registry. This is the ONLY interface between Next.js (Phase 2) and the gbrain CLI — keep it Next.js-agnostic so Phase 2 Route Handlers can import it raw without coupling.

Purpose: gbrain's PGLite store takes an exclusive file lock, so two concurrent spawns against the same brain deadlock at the storage layer. Application-side serialization via a Promise-chain mutex is the surgical fix. Slug validation prevents `GBRAIN_HOME` injection. Typed errors give Route Handlers a clean `try/catch` story.

Output: `lib/gbrain/` module — pure TypeScript, zero Next.js imports, unit-tested via Bun's built-in test runner. Phase 2 imports `{ spawnGBrain, GbrainError, tenantRegistry }` and nothing else.
</objective>

export class GbrainError extends Error {
  readonly stderr: string;
  readonly code: number | null;
  readonly args: readonly string[];
  constructor(message: string, opts: { stderr: string; code: number | null; args: readonly string[] });
}

// lib/gbrain/slug.ts
import { z } from 'zod';
// Strict slug: lowercase letters, digits, hyphens; 1-63 chars; no leading/trailing hyphen.
export const TenantSlugSchema: z.ZodString;
export function parseTenantSlug(input: unknown): string; // throws zod error on invalid

// lib/gbrain/mutex.ts
export function withTenantLock<T>(tenantId: string, fn: () => Promise<T>): Promise<T>;

// lib/gbrain/client.ts
export interface SpawnGBrainOptions {
  tenantId: string;          // required — drives GBRAIN_HOME=./brains/<tenantId>/
  cwd?: string;              // defaults to process.cwd()
  extraEnv?: Record<string, string>;
}
export interface SpawnGBrainResult {
  stdout: string;
  stderr: string;
  code: number | null;
}
export function spawnGBrain(args: readonly string[], opts: SpawnGBrainOptions): Promise<SpawnGBrainResult>;

// lib/gbrain/tenants.ts
export interface TenantRecord {
  id: string;
  brainDir: string; // absolute path to ./brains/<id>/
  createdAt: Date;  // ctime of the dir
}
export class TenantRegistry {
  scan(brainsRoot?: string): TenantRecord[]; // rebuilds the Map from filesystem
  get(id: string): TenantRecord | undefined;
  list(): TenantRecord[];
  add(record: TenantRecord): void;
  remove(id: string): void;
}
export const tenantRegistry: TenantRegistry;
